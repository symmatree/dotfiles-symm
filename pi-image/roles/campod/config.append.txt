# Serial console on GPIO 14/15 -- pins 8 (TXD) and 10 (RXD), ground on pin 6,
# 115200 8N1. The field-debug path on a headless pod: no display, no keyboard.
#
# cmdline.txt already carries console=serial0,115200, but that alone produces
# nothing on a Zero 2 W. The PL011 UART is wired to the Bluetooth modem, so
# GPIO 14/15 get the *mini*-UART -- and the mini-UART is disabled unless
# enable_uart=1. The symptom is total silence, which is indistinguishable from a
# bad solder joint on the header. (Confirmed 2026-09-07: the first campod card
# showed nothing on serial for exactly this reason.)
enable_uart=1

# disable-bt hands the real PL011 back to GPIO 14/15, and here it is necessary
# rather than tidy: the mini-UART derives its baud rate from the VPU core clock,
# and this image sets arm_boost=1, so that clock moves at runtime and mini-UART
# output garbles. Bluetooth is not used on a camera pod.
#
# Leaves the hciuart service with nothing to attach to; it will fail harmlessly.
# Disabling it belongs to the ansible layer, not to the image.
dtoverlay=disable-bt

# USB gadget networking (coordinator#24 Phase 3). The overlay has to come from
# the image: it is device tree, read by the firmware at boot, and nothing in
# userspace can substitute for it. Loading the dwc2/g_ether modules is the
# ansible layer's job (/etc/modules-load.d/), which is why there is no matching
# cmdline.txt change here -- and why this build needs no cmdline knob.
#
# dr_mode=otg rather than peripheral: with a plain micro-B cable (ID floating)
# the controller comes up as a peripheral, which is what g_ether needs to bind;
# with an OTG adapter (ID grounded) it comes up as a host. That covers the pod
# talking to the coordinator AND the pod driving a hub, without this file having
# to encode which one. Force it to "peripheral" or "host" if that flexibility is
# not wanted.
#
# Note the vendor config.txt's existing dtoverlay=dwc2,dr_mode=host line is under
# a [cm5] section and does not apply to a Zero 2 W.
dtoverlay=dwc2,dr_mode=otg
