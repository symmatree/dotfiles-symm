# --- campod (Pi Zero 2 W) role additions ---
# Appended verbatim to the END of the vendor config.txt by build-image.sh, so
# these land in whatever conditional section that file happens to end in. The
# explicit [all] resets the scope: a vendor file ending inside [cm4]/[cm5] would
# make every line below silently inert on a Zero, with no error anywhere.
[all]

# Serial console on GPIO 14/15 -- pins 8 (TXD) and 10 (RXD), ground on pin 6,
# 115200 8N1. The field-debug path on a headless pod: no display, no keyboard.
#
# cmdline.txt already carries console=serial0,115200, but that alone produces
# nothing on a Zero 2 W: the PL011 UART is wired to the Bluetooth modem, so
# GPIO 14/15 get the *mini*-UART, which is disabled unless enable_uart=1. The
# symptom is total silence, indistinguishable from a bad solder joint.
enable_uart=1

# Hands the real PL011 back to GPIO 14/15. Necessary rather than tidy: the
# mini-UART derives its baud rate from the VPU core clock, and arm_boost=1 moves
# that clock at runtime, so mini-UART output garbles. Bluetooth is unused here.
#
# Leaves hciuart with nothing to attach to; it fails harmlessly, and disabling it
# belongs to the ansible layer.
dtoverlay=disable-bt

# USB gadget networking (coordinator#24 Phase 3). Device tree, read by the
# firmware at boot, so it has to come from the image -- nothing in userspace can
# substitute. Loading the dwc2/g_ether modules is the ansible layer's job
# (/etc/modules-load.d/), which is why there is no cmdline knob to match.
#
# dr_mode=peripheral explicitly: four campods hang off a USB hub whose upstream
# goes to the coordinator, so the coordinator is the host and every pod is a
# device on a downstream port, never a host. "otg" would probably land in the
# same place -- ID floating on a plain micro-B cable selects peripheral -- but
# that makes correct behaviour depend on cable wiring across eight units.
#
# The vendor config.txt's own dtoverlay=dwc2,dr_mode=host sits under [cm5] and
# does not apply to a Zero 2 W.
dtoverlay=dwc2,dr_mode=peripheral

# SPI, for the ADXL345 vibration sensors colocated with the camera
# (coordinator#211, #218). Device tree, so it comes from the image for the same
# reason dwc2 does, even though the sensor is payload and its code is not here.
#
# /boot/firmware is FAT -- no checksums, no CoW -- on a card in a vehicle that
# loses power abruptly, and it is the one filesystem the btrfs layout does not
# protect. So the rule is that nothing writes it during normal operation, when a
# power pull can land mid-write and the device has no idea a flight has ended.
# The partition is not immutable: a deliberate operator-initiated write on the
# ground, such as staging a re-flash image (coordinator#310), is the case this
# protects rather than the case it forbids. What has to stay off it is routine
# convergence -- which is why config.txt is built into the image instead of
# edited by ansible on a running device.
#
# Inert with nothing on the bus, so a campod without a sensor pays nothing.
dtparam=spi=on

# CMA. REQUIRED by the vc4-kms-v3d removal in campod.env rather than merely
# paired with it: that overlay is what sets CMA from device tree, and the kernel
# default is CONFIG_CMA_SIZE_MBYTES=5. Remove the overlay without this line and
# the camera has 5MB to allocate from.
#
# 128MB because the overlays README gates cma-256 on 1GB of RAM and this board
# has 512MB. No performance claim attaches to the size.
#
# The camera's actual requirement is measured in coordinator#316 and #339, and a
# too-small reservation presents as a camera that will not start rather than as
# an out-of-memory message -- so the check is that capture works at full sensor
# resolution.
dtoverlay=cma,cma-128

# GPU/CPU memory split. 64 MiB was never chosen on this role -- it is the
# firmware default for a board under 1 GB -- and this device has no display, no
# vc4-kms-v3d, drm blacklisted, no cards in /sys/class/drm and "no soundcards"
# in /proc/asound.
#
# 32 is the smallest usable value on this board. Measured on campod-se, one
# variable at a time:
#
#   set   reported   firmware   VCHI        ISP nodes   camera
#   16    gpu=16M    start_cd   fails -22   absent      rpicam refuses
#   17    gpu=16M    start_cd   fails -22   absent      rpicam refuses
#   24    gpu=16M    start_cd   fails -22   absent      rpicam refuses
#   32    gpu=32M    full       ok          video13-16  enumerates 4608x2592
#   64    gpu=64M    full       ok          present     works
#
# TWO THINGS THAT COST A BREAKAGE TO LEARN, neither of them documented.
#
# The firmware rounds gpu_mem DOWN TO A MULTIPLE OF 16. 17 and 24 both report
# gpu=16M. The documentation says the cut-down firmware is selected by
# "gpu_mem=16", which reads as an exact match; in practice anything that rounds
# to 16 selects it. There is no published granularity and start.elf is closed.
#
# The cut-down firmware is incompatible with this camera. It is not a size
# problem. start_cd does not provide the VideoCore services the ISP rides on:
#
#   vc_sm_cma_vchi_init: failed to open VCHI service (-22)
#   bcm2835_mmal_vchiq: Failed to open VCHI service connection (status=-22)
#
# bcm2835_isp then creates no device nodes -- /dev/video13..16 are simply absent
# -- and libcamera's vc4 pipeline drives the ISP through exactly that path. The
# sensor is fine throughout (imx708 probes, module ID 0x0302, over I2C) and so is
# Unicam, which is what makes this fail as "camera will not start" rather than as
# anything that names memory.
#
# So the 16 MiB that only the cut-down firmware offers is not available to us at
# any price, and 32 returns 31.9 MiB -- MemTotal 424728 -> 457432 kB -- against
# 161.9 MiB of page cache, which is the only reclaimable memory on this board
# (coordinator#434, coordinator#447).
#
# The check remains #316's: capture works at full sensor resolution. A value that
# is too small, or a missing firmware component, fails inside libcamera and
# presents as a camera that will not start.
gpu_mem=32

# PPS input: this pod's buffered 1 Hz leg from the distribution board, on BCM
# GPIO26 = header pin 37, with pin 39 as its signal ground (coordinator#493,
# docs/campod.md). Gives /dev/pps0 for chrony, which takes the fraction of a
# second from here and the integer second from the coordinator over the gadget
# network -- coordinator#370.
#
# A 1x2 in the odd row at the far end of the header, and the constraint is
# mechanical rather than electrical: PPS needs a pair of its own that no other
# connector sits on top of. It keeps away from the accelerometer's 2x5 at header
# pins 17-26 and the serial console at 6/8/10, so none of the three can foul
# another. Header pins 9/11 were the other candidate and are out -- they sit
# alongside the console plug.
#
# SO "SPARE" IN A PIN TABLE DOES NOT MEAN AVAILABLE. An earlier version of this
# line used GPIO24 on header 18, justified from docs/campod-software.md listing
# that position as "spare -- INT". It is spare in the sense that nothing is
# assigned to it, and unavailable in the sense that the accelerometer's 2x5
# covers header 17-26 whole -- a tenth position inside an occupied block is
# reachable through THAT connector and by nothing else. Header 18 and 22 are
# still the right spares for an accelerometer interrupt, on the accelerometer's
# own cable; they were never candidates for a second connector.
#
# NOT the coordinator's pin, which is GPIO24 on header 18. There the signal
# arrives as one line of a 2x4 that also carries SPI and power for the DS3234; a
# pod has no SPI connection to the chip and takes only the buffered edge, so the
# two roles differ in pin and agree in edge. That one buffered copy of a single
# SQW line is what makes pods share phase by construction rather than by
# agreement.
#
# Header position and BCM number are different numbering schemes and both appear
# above: header 37 is BCM 26, while header 26 is BCM 7 -- CE1, the arm-end
# accelerometer's chip select.
#
# assert_falling_edge=1, the same as the coordinator and for the same reason: the
# falling edge is the top of the second and pps-gpio defaults to the rising one.
# The mechanism is in roles/coordinator/config.append.txt beside its own
# pps-gpio line -- one copy, not two. A pod timing a different edge from the
# coordinator would put a 500 ms step between them, which is the one thing the
# shared wire exists to rule out.
#
# Harmless before a pod's leg is fitted: pps-gpio on an unconnected pin presents
# /dev/pps0 and no pulses, which is a quiet device rather than an error.
dtoverlay=pps-gpio,gpiopin=26,assert_falling_edge=1
