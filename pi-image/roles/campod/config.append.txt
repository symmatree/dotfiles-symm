# --- campod (Pi Zero 2 W) role additions ---
# Appended verbatim to the END of the vendor config.txt by build-image.sh, so
# these land in whatever conditional section that file happens to end in. The
# explicit [all] resets the scope: a vendor file ending inside [cm4]/[cm5] would
# make every line below silently inert on a Zero, with no error anywhere.
[all]

# Serial console on GPIO 14/15 -- pins 8 (TXD) and 10 (RXD), ground on pin 6,
# 115200 8N1. The field-debug path on a headless pod: no display, no keyboard.
#
# cmdline.txt already carries console=serial0,115200, but that alone produces
# nothing on a Zero 2 W: the PL011 UART is wired to the Bluetooth modem, so
# GPIO 14/15 get the *mini*-UART, which is disabled unless enable_uart=1. The
# symptom is total silence, indistinguishable from a bad solder joint.
enable_uart=1

# Hands the real PL011 back to GPIO 14/15. Necessary rather than tidy: the
# mini-UART derives its baud rate from the VPU core clock, and arm_boost=1 moves
# that clock at runtime, so mini-UART output garbles. Bluetooth is unused here.
#
# Leaves hciuart with nothing to attach to; it fails harmlessly, and disabling it
# belongs to the ansible layer.
dtoverlay=disable-bt

# USB gadget networking (coordinator#24 Phase 3). Device tree, read by the
# firmware at boot, so it has to come from the image -- nothing in userspace can
# substitute. Loading the dwc2/g_ether modules is the ansible layer's job
# (/etc/modules-load.d/), which is why there is no cmdline knob to match.
#
# dr_mode=peripheral explicitly: four campods hang off a USB hub whose upstream
# goes to the coordinator, so the coordinator is the host and every pod is a
# device on a downstream port, never a host. "otg" would probably land in the
# same place -- ID floating on a plain micro-B cable selects peripheral -- but
# that makes correct behaviour depend on cable wiring across eight units.
#
# The vendor config.txt's own dtoverlay=dwc2,dr_mode=host sits under [cm5] and
# does not apply to a Zero 2 W.
dtoverlay=dwc2,dr_mode=peripheral

# SPI, for the ADXL345 vibration sensors colocated with the camera
# (coordinator#211, #218). Device tree, so it comes from the image for the same
# reason dwc2 does, even though the sensor is payload and its code is not here.
#
# /boot/firmware is FAT -- no checksums, no CoW -- on a card in a vehicle that
# loses power abruptly, and it is the one filesystem the btrfs layout does not
# protect. So the rule is that nothing writes it during normal operation, when a
# power pull can land mid-write and the device has no idea a flight has ended.
# The partition is not immutable: a deliberate operator-initiated write on the
# ground, such as staging a re-flash image (coordinator#310), is the case this
# protects rather than the case it forbids. What has to stay off it is routine
# convergence -- which is why config.txt is built into the image instead of
# edited by ansible on a running device.
#
# Inert with nothing on the bus, so a campod without a sensor pays nothing.
dtparam=spi=on

# CMA. This line is REQUIRED by the vc4-kms-v3d removal in campod.env, not
# merely paired with it: that overlay is what device tree uses to set CMA, and
# the kernel's own default is CONFIG_CMA_SIZE_MBYTES=5. Remove the overlay
# without this and CMA is 5MB, under the camera's ~48 MiB, and capture stops.
#
# The overlays README warns that this overlay "might clash with other overlays
# like vc4-fkms-v3d and vc4-kms-v3d" -- which is precisely why it only appears
# on a role that removes vc4-kms-v3d.
#
# 128MB against a measured requirement of ~48 MiB. capture.py runs
# create_still_configuration at sensor resolution with CAMPOD_CAPTURE_WIDTH and
# HEIGHT unset, which in picamera2 means buffer_count=1, a BGR888 main stream and
# a raw stream at the same size. For the IMX708 at 4608x2592 that is 34.2 MiB of
# main plus 14.2 MiB of raw, plus small ISP statistics buffers.
#
# The number is tied to that configuration, not to the camera: a video
# configuration defaults to six buffers, which at this resolution would want
# ~290 MiB and fits in none of these values. If CAMPOD_CAPTURE_WIDTH/HEIGHT or
# the configuration type changes, redo the arithmetic.
#
# A too-small reservation fails inside libcamera's buffer allocation, which
# presents as a camera that will not start rather than as an out-of-memory
# message. The check is that capture works at full sensor resolution.
dtoverlay=cma,cma-128

# GPU/CPU memory split. 64 MiB was never chosen -- it is the firmware's default
# for a board with less than 1 GB, and on this role it is 64 MiB withheld from a
# device that has no display, no vc4-kms-v3d, drm blacklisted, no cards in
# /sys/class/drm and "no soundcards" in /proc/asound.
#
# 16 is the documented minimum. Raspberry Pi's config.txt documentation, on the
# question that decides it:
#
#   "The camera stack (libcamera) on Raspberry Pi OS uses Linux CMA memory to
#    allocate buffers instead of GPU memory, so there is no benefit in
#    increasing the GPU memory size."
#
# So the camera's buffers come from the CMA region set above, not from here.
# Measured on campod-se, MemTotal was 424728 kB with gpu=64M; this should return
# roughly 48 MiB to the ARM, against 161.9 MiB of page cache which is the only
# reclaimable memory on the board (coordinator#434).
#
# 16 IS NOT JUST A SMALLER SPLIT. It is the only value that selects the cut-down
# firmware, and the same documentation is explicit about what that gives up:
#
#   "Cut-down firmware (start*cd.elf and fixup*cd.dat) cannot be selected this
#    way - the system will fail to boot. The only way to enable the cut-down
#    firmware is to specify gpu_mem=16. The cut-down firmware removes support
#    for codecs, 3D and debug logging as well as limiting the initial early-boot
#    framebuffer to 1080p @16bpp [...]"
#
# Codecs, 3D and an early-boot framebuffer are all things this role has already
# given up by other means: vc4-kms-v3d is removed, drm is blacklisted, there is
# no display, and nothing here encodes video. So the losses line up with the
# role rather than being a price paid for the memory.
#
# It also makes the bcm2835_codec entry in campod.env's MODULE_BLACKLIST a
# consequence rather than a guess: with cut-down firmware there is no codec
# behind that module. The two changes are coupled, not independent.
#
# Alpine packages this firmware as raspberrypi-bootloader-cutdown, so it is a
# configuration other distributions ship rather than an exotic one, and nothing
# in it is flagged as camera-affecting.
#
# If capture does break, bcm2835_isp is bound to bcm2835_mmal_vchiq -- the ISP is
# reached through MMAL over the VideoCore mailbox -- so the firmware-side half of
# the camera path is the first place to look. That is where to start, not a
# reason to hesitate: the check is #316's, capture works at full sensor
# resolution, and a config.txt line on a mounted FAT partition is one edit and a
# reboot to back out.
gpu_mem=16. The cut-down firmware removes support
#    for codecs, 3D and debug logging as well as limiting the initial early-boot
#    framebuffer to 1080p @16bpp [...]"
#
# Codecs, 3D and an early-boot framebuffer are all things this role has already
# given up by other means: vc4-kms-v3d is removed, drm is blacklisted, there is
# no display, and nothing here encodes video. So the losses line up with the
# role rather than being a price paid for the memory.
#
# It also makes the bcm2835_codec entry in campod.env's MODULE_BLACKLIST a
# consequence rather than a guess: with cut-down firmware there is no codec
# behind that module. The two changes are coupled, not independent.
#
# WHAT WOULD DISPROVE IT, and it is one specific thing. bcm2835_isp is bound to
# bcm2835_mmal_vchiq -- the ISP is reached through MMAL over the VideoCore
# mailbox -- and vc_sm_cma shares memory between ARM and VideoCore. If the ISP's
# firmware-side half is part of what "codecs" drops, the camera stops working.
# The documentation does not say either way, and buffers coming from CMA does
# not answer it, because firmware-side processing is not a buffer.
#
# So the check is #316's and it is load-bearing here: capture works at full
# sensor resolution. A too-small reservation, or a missing firmware component,
# fails inside libcamera and presents as a camera that will not start rather
# than as an out-of-memory message. config.txt is an ordinary file on a mounted
# FAT partition, so backing this out costs one edit and a reboot, not a reflash.
gpu_mem=16
